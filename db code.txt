Table roles {
  id   int          [pk, increment]
  name varchar(20)  [unique, not null] // admin, manager
}

Table users {
  id            int           [pk, increment, not null]
  name          varchar(100)  [not null]
  email         varchar(150)  [unique, not null]
  password_hash varchar(255)  [not null]
  role_id       int           [not null, ref: > roles.id]
  is_active     boolean       [not null, default: true]
  created_at    timestamp     [not null, default: `now()`]
  updated_at    timestamp     [not null, default: `now()`]

  indexes {
    email   [unique, name: 'uq_users_email']
    role_id [name: 'idx_users_role_id']
  }
}

Table categories {
  id          int          [pk, increment, not null]
  name        varchar(100) [unique, not null]
  description text
  created_at  timestamp    [not null, default: `now()`]

  indexes {
    name [unique, name: 'uq_categories_name']
  }
}

Table suppliers {
  id           int          [pk, increment, not null]
  name         varchar(150) [not null]
  contact_name varchar(100)
  email        varchar(150) [unique]
  phone        varchar(20)
  address      text
  created_at   timestamp    [not null, default: `now()`]

  indexes {
    email [name: 'idx_suppliers_email']
  }
}

Table products {
  id              int            [pk, increment, not null]
  productcode     varchar(50)    [unique, not null]
  name            varchar(150)   [not null]
  description     text
  category_id     int            [not null, ref: > categories.id]
  supplier_id     int            [ref: > suppliers.id, note: 'nullable — supplier is optional']
  price           decimal(10, 2) [not null, default: 0, note: 'must be >= 0']
  quantity        int            [not null, default: 0, note: 'must be >= 0']
  min_stock_level int            [not null, default: 10, note: 'triggers low stock alert']
  status          varchar(20)    [not null, default: 'active', note: 'active | inactive | discontinued']
  created_at      timestamp      [not null, default: `now()`]
  updated_at      timestamp      [not null, default: `now()`]

  indexes {
    productcode         [unique, name: 'uq_products_sku']
    category_id         [name: 'idx_products_category']
    supplier_id         [name: 'idx_products_supplier']
    status              [name: 'idx_products_status']
    (name, category_id) [name: 'idx_products_name_category']
  }
}

Table stock_movements {
  id                int         [pk, increment, not null]
  product_id        int         [not null, ref: > products.id]
  user_id           int         [not null, ref: > users.id]
  movement_type     varchar(10) [not null, note: 'IN | OUT | ADJUST']
  quantity_changed  int         [not null, note: 'positive for IN, negative for OUT']
  previous_quantity int         [not null]
  new_quantity      int         [not null]
  reference_no      varchar(50) [note: 'optional PO / invoice number']
  notes             text
  created_at        timestamp   [not null, default: `now()`]

  indexes {
    product_id               [name: 'idx_movements_product']
    user_id                  [name: 'idx_movements_user']
    movement_type            [name: 'idx_movements_type']
    created_at               [name: 'idx_movements_date']
    (product_id, created_at) [name: 'idx_movements_product_date']
  }
}

Table alerts {
  id          int         [pk, increment, not null]
  product_id  int         [not null, ref: > products.id]
  alert_type  varchar(30) [not null, note: 'LOW_STOCK | OUT_OF_STOCK | EXPIRY']
  message     text        [not null]
  is_resolved boolean     [not null, default: false]
  resolved_by int         [ref: > users.id, note: 'nullable — set when resolved']
  resolved_at timestamp   [note: 'nullable — set when resolved']
  created_at  timestamp   [not null, default: `now()`]

  indexes {
    product_id                [name: 'idx_alerts_product']
    is_resolved               [name: 'idx_alerts_resolved']
    alert_type                [name: 'idx_alerts_type']
    (product_id, is_resolved) [name: 'idx_alerts_product_resolved']
  }
}
